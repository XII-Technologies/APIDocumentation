30 inline xiiResult Dispatch(xiiLightClusterConstants& snapshot, xiiUInt32 count, xiiGALDispatchComputeDescription& out_dispatch)
31 {
32 if (!xiiGALDevice::HasDefaultDevice() || !count)
33 return XII_FAILURE;
34 const auto& limits = xiiGALDevice::GetDefaultDevice()->GetGraphicsDeviceAdapterProperties().m_ComputeShaderProperties;
35 if (limits.m_uiMaxThreadGroupInvocations < 64 || limits.m_uiMaxThreadGroupSizeX < 64 || !limits.m_uiMaxThreadGroupCountX)
36 return XII_FAILURE;
37 const xiiUInt32 groups = (count + 63U) / 64U;
38 out_dispatch.m_uiThreadGroupCountX = xiiMath::Min(groups, xiiMath::Min(limits.m_uiMaxThreadGroupCountX, xiiMath::MaxValue<xiiUInt32>() / 64U));
39 out_dispatch.m_uiThreadGroupCountY = (groups + out_dispatch.m_uiThreadGroupCountX - 1U) / out_dispatch.m_uiThreadGroupCountX;
40 out_dispatch.m_uiThreadGroupCountZ = 1;
41 snapshot.Work.z = out_dispatch.m_uiThreadGroupCountX * 64U;
42 return out_dispatch.m_uiThreadGroupCountY <= limits.m_uiMaxThreadGroupCountY ? XII_SUCCESS : XII_FAILURE;
43 }
44 inline void Bind(xiiRenderGraphPassContext& execute, xiiRenderGraphBufferHandle handle, const xiiLightClusterConstants& snapshot, xiiGALComputePipelineState* pipeline)
45 {
46 auto& commands = execute.GetCommandList();
47 {
48 xiiGALMapHelper<xiiLightClusterConstants> mapped(commands, execute.GetBuffer(handle), xiiGALMapType::Write, xiiGALMapFlags::Discard);
49 *mapped = snapshot;
50 }
51 xiiGALStateTransitionDescription transition;
52 transition.m_pResource = execute.GetBuffer(handle);
53 transition.m_NewState = xiiGALResourceStateFlags::ConstantBuffer;
54 transition.m_TransitionFlags = xiiGALStateTransitionFlags::UpdateState;
55 commands.TransitionResourceStates(xiiMakeArrayPtr(&transition, 1));
56 commands.SetPipelineState(pipeline);
57 commands.ResolveAndSetConstantBuffer("xiiLightClusterConstants", execute.GetBuffer(handle), xiiGALShaderType::Compute);
58 }
