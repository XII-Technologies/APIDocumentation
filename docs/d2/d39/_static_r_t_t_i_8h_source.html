88 template <typename Type>
89 xiiBitflags<xiiTypeFlags> DetermineTypeFlags()
90 {
91 xiiBitflags<xiiTypeFlags> flags;
92 xiiVariantType::Enum type =
93 static_cast<xiiVariantType::Enum>(xiiVariantTypeDeduction<typename xiiTypeTraits<Type>::NonConstReferenceType>::value);
94 if ((type >= xiiVariantType::FirstStandardType && type <= xiiVariantType::LastStandardType) || XII_IS_SAME_TYPE(xiiVariant, Type))
95 flags.Add(xiiTypeFlags::StandardType);
96 else
97 flags.Add(xiiTypeFlags::Class);
98
99 if (std::is_abstract<Type>::value)
100 flags.Add(xiiTypeFlags::Abstract);
101
102 return flags;
103 }
104
105 template <>
106 XII_ALWAYS_INLINE xiiBitflags<xiiTypeFlags> DetermineTypeFlags<xiiVariant>()
107 {
108 return xiiTypeFlags::StandardType;
109 }
