44 struct Scheduling
45 {
46 [[nodiscard]] static XII_ALWAYS_INLINE xiiUInt32 GetQueueIndex(xiiBitflags<xiiGALCommandQueueFlags> flags, bool bAsyncQueuesEnabled)
47 {
48 if (!bAsyncQueuesEnabled)
49 return 0U;
50
51 // Queue capability values are hierarchical: Graphics contains Compute and Transfer,
52 // and Compute contains Transfer. IsSet() tests for any overlapping bit and therefore
53 // classifies every non-empty queue mask as graphics. Test complete capabilities from
54 // most to least capable instead.
55 if (flags.AreAllSet(xiiGALCommandQueueFlags::Graphics))
56 return 0U;
57 if (flags.AreAllSet(xiiGALCommandQueueFlags::Compute))
58 return 1U;
59 if (flags.AreAllSet(xiiGALCommandQueueFlags::Transfer))
60 return 2U;
61 return 0U;
62 }
63 };
