9 {
10 const auto* d = context.m_Graph.GetTextureDescription(image);
11 if (!d || !d->m_Size.HasNonZeroArea() || (size && d->m_Size != *size) || d->m_Type != xiiGALResourceDimension::Texture2D || d->m_uiArraySizeOrDepth != 1 || d->m_uiSampleCount != 1 || !d->m_BindFlags.IsSet(xiiGALBindFlags::ShaderResource) || xiiGALTextureUtilities::GetComponentCount(d->m_Format) < channels || (channels > 1 && xiiGALResourceFormat::IsDepthFormat(d->m_Format)))
12 return nullptr;
13 const auto& p = xiiGALTextureUtilities::GetResourceFormatProperties(d->m_Format);
14 return !p.IsTypeless() && (p.m_ComponentType == xiiGALResourceFormatComponentType::Float || p.IsNormalized() || p.m_ComponentType == xiiGALResourceFormatComponentType::UnsignedNormalizedSRGB || p.m_ComponentType == xiiGALResourceFormatComponentType::Depth || p.m_ComponentType == xiiGALResourceFormatComponentType::DepthStencil) ? d : nullptr;
15 }
