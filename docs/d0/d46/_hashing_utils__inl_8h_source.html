46struct xiiHashHelper<xiiUInt8>
47{
48 XII_ALWAYS_INLINE static xiiUInt32 Hash(xiiUInt8 value)
49 {
50 // Promote to 32-bit then run a small, fast finalizer (Murmur3 / splitmix style).
51 xiiUInt32 z = xiiUInt32(value);
52 z += 0x9E3779B9U;
53 z = (z ^ (z >> 16)) * 0x85EBCA6BU;
54 z = (z ^ (z >> 13)) * 0xc2B2AE35U;
55 return z ^ (z >> 16);
56 }
57
58 XII_ALWAYS_INLINE static bool Equal(xiiUInt8 a, xiiUInt8 b) { return a == b; }
59};

105 // Knuth: multiplication by the golden ratio will minimize gaps in the hash space.
