116template <xiiUInt32 BlockSize>
117void* xiiLargeBlockAllocator<BlockSize>::Allocate(size_t uiAlign)
118{
119 XII_ASSERT_RELEASE(xiiMath::IsPowerOf2((xiiUInt32)uiAlign), "Alignment must be power of two");
120
121 xiiTime fAllocationTime = xiiTime::Now();
122
123 XII_LOCK(m_Mutex);
124
125 void* ptr = nullptr;
126
127 if (!m_FreeBlocks.IsEmpty())
128 {
129 // Re-use a super block
130 xiiUInt32 uiFreeBlockIndex = m_FreeBlocks.PeekBack();
131 m_FreeBlocks.PopBack();
132
133 const xiiUInt32 uiSuperBlockIndex = uiFreeBlockIndex / SuperBlock::NUM_BLOCKS;
134 const xiiUInt32 uiInnerBlockIndex = uiFreeBlockIndex & (SuperBlock::NUM_BLOCKS - 1);
135 SuperBlock& superBlock = m_SuperBlocks[uiSuperBlockIndex];
136 ++superBlock.m_uiUsedBlocks;
137
138 ptr = xiiMemoryUtils::AddByteOffset(superBlock.m_pBasePtr, uiInnerBlockIndex * BlockSize);
139 }
140 else
141 {
142 // Allocate a new super block
143 void* pMemory = xiiPageAllocator::AllocatePage(SuperBlock::SIZE_IN_BYTES);
144 XII_CHECK_ALIGNMENT(pMemory, uiAlign);
145
146 SuperBlock superBlock;
147 superBlock.m_pBasePtr = pMemory;
148 superBlock.m_uiUsedBlocks = 1;
149
150 m_SuperBlocks.PushBack(superBlock);
151
152 const xiiUInt32 uiBlockBaseIndex = (m_SuperBlocks.GetCount() - 1) * SuperBlock::NUM_BLOCKS;
153 for (xiiUInt32 i = SuperBlock::NUM_BLOCKS - 1; i > 0; --i)
154 {
155 m_FreeBlocks.PushBack(uiBlockBaseIndex + i);
156 }
157
158 ptr = pMemory;
159 }
160
161 if (m_TrackingMode >= xiiAllocatorTrackingMode::AllocationStats)
162 {
163 xiiMemoryTracker::AddAllocation(m_Id, m_TrackingMode, ptr, BlockSize, uiAlign, xiiTime::Now() - fAllocationTime);
164 }
165
166 return ptr;
167}

169template <xiiUInt32 BlockSize>
170void xiiLargeBlockAllocator<BlockSize>::Deallocate(void* pPtr)
171{
172 XII_LOCK(m_Mutex);
173
174 if (m_TrackingMode >= xiiAllocatorTrackingMode::AllocationStats)
175 {
176 xiiMemoryTracker::RemoveAllocation(m_Id, pPtr);
177 }
178
179 // find super block
180 bool bFound = false;
181 xiiUInt32 uiSuperBlockIndex = m_SuperBlocks.GetCount();
182 std::ptrdiff_t uiPtrDiff = 0;
183
184 for (; uiSuperBlockIndex-- > 0;)
185 {
186 uiPtrDiff = (char*)pPtr - (char*)m_SuperBlocks[uiSuperBlockIndex].m_pBasePtr;
187 if (uiPtrDiff >= 0 && uiPtrDiff < SuperBlock::SIZE_IN_BYTES)
188 {
189 bFound = true;
190 break;
191 }
192 }
193
194 XII_IGNORE_UNUSED(bFound);
195 XII_ASSERT_DEV(bFound, "'{0}' was not allocated with this allocator.", xiiArgP(pPtr));
196
197 SuperBlock& superBlock = m_SuperBlocks[uiSuperBlockIndex];
198 --superBlock.m_uiUsedBlocks;
199
200 if (superBlock.m_uiUsedBlocks == 0 && m_FreeBlocks.GetCount() > SuperBlock::NUM_BLOCKS * 4)
201 {
202 // Give memory back.
203 xiiPageAllocator::DeallocatePage(superBlock.m_pBasePtr);
204
205 m_SuperBlocks.RemoveAtAndSwap(uiSuperBlockIndex);
206 const xiiUInt32 uiLastSuperBlockIndex = m_SuperBlocks.GetCount();
207
208 // Patch free list.
209 for (xiiUInt32 i = 0; i < m_FreeBlocks.GetCount(); ++i)
210 {
211 const xiiUInt32 uiIndex = m_FreeBlocks[i];
212 const xiiUInt32 uiSBIndex = uiIndex / SuperBlock::NUM_BLOCKS;
213
214 if (uiSBIndex == uiSuperBlockIndex)
215 {
216 // Points to the block we just removed.
217 m_FreeBlocks.RemoveAtAndSwap(i);
218 --i;
219 }
220 else if (uiSBIndex == uiLastSuperBlockIndex)
221 {
222 // Points to the block we just swapped.
223 m_FreeBlocks[i] = uiSuperBlockIndex * SuperBlock::NUM_BLOCKS + (uiIndex & (SuperBlock::NUM_BLOCKS - 1));
224 }
225 }
226 }
227 else
228 {
229 // Add block to free list.
230 const xiiUInt32 uiInnerBlockIndex = (xiiUInt32)(uiPtrDiff / BlockSize);
231 m_FreeBlocks.PushBack(uiSuperBlockIndex * SuperBlock::NUM_BLOCKS + uiInnerBlockIndex);
232 }
233}
