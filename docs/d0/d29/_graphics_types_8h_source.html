957 if (uiPrimitiveCount == 0)
958 return 0;
959
960 if (e >= xiiGALPrimitiveTopology::ControlPointPatchList1 && e <= xiiGALPrimitiveTopology::ControlPointPatchList32)
961 {
962 xiiUInt32 uiVerticesPerPrimitive = VerticesPerPrimitive(e);
963 return uiPrimitiveCount * uiVerticesPerPrimitive; // Each primitive requires 'uiVerticesPerPrimitive' indices.
964 }
965
966 switch (e)
967 {
968 case xiiGALPrimitiveTopology::PointList:
969 return uiPrimitiveCount; // 1 index per primitive.
970
971 case xiiGALPrimitiveTopology::LineList:
972 case xiiGALPrimitiveTopology::LineStrip:
973 return uiPrimitiveCount * 2; // 2 indices per primitive.
974
975 case xiiGALPrimitiveTopology::TriangleList:
976 return uiPrimitiveCount * 3; // 3 indices per primitive.
977
978 case xiiGALPrimitiveTopology::TriangleStrip:
979 return uiPrimitiveCount + 2; // First two indices are shared.
980
981 case xiiGALPrimitiveTopology::TriangleListAdjacent:
982 return uiPrimitiveCount * 6; // 6 indices per primitive.
983
984 case xiiGALPrimitiveTopology::TriangleStripAdjacent:
985 return uiPrimitiveCount + 4; // Adjusted for adjacency.
986
987 case xiiGALPrimitiveTopology::LineListAdjacent:
988 case xiiGALPrimitiveTopology::LineStripAdjacent:
989 return uiPrimitiveCount * 4; // 4 indices per primitive.
990
991 default:
992 return 0;
993 }
