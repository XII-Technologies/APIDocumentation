47inline xiiColor xiiColor::MakeFromKelvin(xiiUInt32 uiKelvin)
48{
49 xiiColor finalColor;
50 float fKelvin = xiiMath::Clamp(uiKelvin, 1000u, 40000u) / 1000.0f;
51 float fKelvinSquared = fKelvin * fKelvin;
52
53 finalColor.r = fKelvin < 6.570f ? 1.0f : xiiMath::Saturate((1.35651f + 0.216422f * fKelvin + 0.000633715f * fKelvinSquared) / (-3.24223f + 0.918711f * fKelvin));
54 finalColor.g = fKelvin < 6.570f ? xiiMath::Saturate((-399.809f + 414.271f * fKelvin + 111.543f * fKelvinSquared) / (2779.24f + 164.143f * fKelvin + 84.7356f * fKelvinSquared)) : xiiMath::Saturate((1370.38f + 734.616f * fKelvin + 0.689955f * fKelvinSquared) / (-4625.69f + 1699.87f * fKelvin));
55 finalColor.b = fKelvin > 6.570f ? 1.0f : xiiMath::Saturate((348.963f - 523.53f * fKelvin + 183.62f * fKelvinSquared) / (2848.82f - 214.52f * fKelvin + 78.8614f * fKelvinSquared));
56
57 return finalColor;
58}
