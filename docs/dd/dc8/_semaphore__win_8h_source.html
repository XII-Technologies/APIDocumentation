24xiiResult xiiSemaphore::Create(xiiUInt32 uiInitialTokenCount, xiiStringView sSharedName /*= nullptr*/)
25{
26 XII_ASSERT_DEV(m_hSemaphore == nullptr, "Semaphore can't be recreated.");
27
28 LPSECURITY_ATTRIBUTES pSecurityAttributes = nullptr; // Default.
29 const DWORD uiFlags = 0; // Reserved but unused.
30 const DWORD uiAccess = STANDARD_RIGHTS_ALL | SEMAPHORE_MODIFY_STATE /* needed for ReleaseSemaphore */;
31
32 if (sSharedName.IsEmpty())
33 {
34 // Create an unnamed semaphore.
35
36 m_hSemaphore = CreateSemaphoreExW(pSecurityAttributes, uiInitialTokenCount, xiiMath::MaxValue<xiiInt32>(), nullptr, uiFlags, uiAccess);
37 }
38 else
39 {
40 // Create a named semaphore in the 'Local' namespace.
41 // These are visible session wide, ie. all processes by the same user account can see these, but not across users.
42
43 const xiiStringBuilder sSemaphoreName("Local\\", sSharedName);
44
45 m_hSemaphore = CreateSemaphoreExW(pSecurityAttributes, uiInitialTokenCount, xiiMath::MaxValue<xiiInt32>(), xiiStringWChar(sSemaphoreName).GetData(), uiFlags, uiAccess);
46 }
47
48 if (m_hSemaphore == nullptr)
49 {
50 return XII_FAILURE;
51 }
52
53 return XII_SUCCESS;
54}

90{
