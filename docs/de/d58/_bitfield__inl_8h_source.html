61template <class Container>
62bool xiiBitfield<Container>::IsAnyBitSet(xiiUInt32 uiFirstBit /*= 0*/, xiiUInt32 uiNumBits /*= 0xFFFFFFFF*/) const
63{
64 if (m_uiCount == 0 || uiNumBits == 0)
65 return false;
66
67 XII_ASSERT_DEBUG(uiFirstBit < m_uiCount, "Cannot access bit {0}, the bitfield only has {1} bits.", uiFirstBit, m_uiCount);
68
69 const xiiUInt32 uiLastBit = xiiMath::Min<xiiUInt32>(uiFirstBit + uiNumBits, m_uiCount) - 1;
70
71 const xiiUInt32 uiFirstInt = GetBitInt(uiFirstBit);
72 const xiiUInt32 uiLastInt = GetBitInt(uiLastBit);
73
74 // all within the same int
75 if (uiFirstInt == uiLastInt)
76 {
77 for (xiiUInt32 i = uiFirstBit; i <= uiLastBit; ++i)
78 {
79 if (IsBitSet(i))
80 return true;
81 }
82 }
83 else
84 {
85 const xiiUInt32 uiNextIntBit = (uiFirstInt + 1) * 32;
86 const xiiUInt32 uiPrevIntBit = uiLastInt * 32;
87
88 // check the bits in the first int individually
89 for (xiiUInt32 i = uiFirstBit; i < uiNextIntBit; ++i)
90 {
91 if (IsBitSet(i))
92 return true;
93 }
94
95 // check the bits in the ints in between with one operation
96 for (xiiUInt32 i = uiFirstInt + 1; i < uiLastInt; ++i)
97 {
98 if ((m_Container[i] & 0xFFFFFFFF) != 0)
99 return true;
100 }
101
102 // check the bits in the last int individually
103 for (xiiUInt32 i = uiPrevIntBit; i <= uiLastBit; ++i)
104 {
105 if (IsBitSet(i))
106 return true;
107 }
108 }
109
110 return false;
111}

447template <class Container>
448void xiiBitfield<Container>::ConstIterator::FindNextChunk(xiiUInt32 uiStartChunk)
449{
450 if (uiStartChunk < m_pBitfield->m_Container.GetCount())
451 {
452 const xiiUInt32 uiLastChunk = m_pBitfield->m_Container.GetCount() - 1;
453 for (xiiUInt32 i = uiStartChunk; i < uiLastChunk; ++i)
454 {
455 if (m_pBitfield->m_Container[i] != 0)
456 {
457 m_uiChunk = i;
458 m_Iterator = sub_iterator(m_pBitfield->m_Container[i]);
459 return;
460 }
461 }
462
463 const xiiUInt32 uiMask = 0xFFFFFFFF >> (32 - (m_pBitfield->m_uiCount - (uiLastChunk << 5)));
464 if ((m_pBitfield->m_Container[uiLastChunk] & uiMask) != 0)
465 {
466 m_uiChunk = uiLastChunk;
467 m_Iterator = sub_iterator(m_pBitfield->m_Container[uiLastChunk] & uiMask);
468 return;
469 }
470 }
471
472 // End iterator.
473 m_pBitfield = nullptr;
474 m_uiChunk = 0;
475 m_Iterator = sub_iterator();
476}
