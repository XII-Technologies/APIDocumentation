290struct xiiRTTIDefaultAllocator : public xiiRTTIAllocator

293 virtual xiiInternal::NewInstance<void> AllocateInternal(xiiAllocator* pAllocator) override // [tested]
294 {
295 if (pAllocator == nullptr)
296 {
297 pAllocator = AllocatorWrapper::GetAllocator();
298 }
299
300 return XII_NEW(pAllocator, CLASS);
301 }
302
304 virtual xiiInternal::NewInstance<void> CloneInternal(const void* pObject, xiiAllocator* pAllocator) override // [tested]
305 {
306 if (pAllocator == nullptr)
307 {
308 pAllocator = AllocatorWrapper::GetAllocator();
309 }
310
311 if constexpr (std::is_copy_constructible_v<CLASS>)
312 {
313 return XII_NEW(pAllocator, CLASS, *static_cast<const CLASS*>(pObject));
314 }
315 else
316 {
317 XII_REPORT_FAILURE("Clone failed since the type is not copy constructible");
318 return xiiInternal::NewInstance<void>(nullptr, pAllocator);
319 }
320 }
321
323 virtual void Deallocate(void* pObject, xiiAllocator* pAllocator) override // [tested]
324 {
325 if (pAllocator == nullptr)
326 {
327 pAllocator = AllocatorWrapper::GetAllocator();
328 }
329
330 CLASS* pPointer = static_cast<CLASS*>(pObject);
331 XII_DELETE(pAllocator, pPointer);
332 }
