210 {
211 if (m_pCurPtr != nullptr && m_pStartPtr < m_pCurPtr)
212 xiiUnicodeUtils::MoveToPriorUtf8(m_pCurPtr, m_pStartPtr).AssertSuccess();
213 else

217 }

228 m_pCurPtr = szOldPos;
229 }

234 }

236 }
