102inline void xiiSimdBBoxSphere::ExpandToInclude(const xiiSimdBBoxSphere& rhs)
103{
104 xiiSimdBBox box = GetBox();
105 box.ExpandToInclude(rhs.GetBox());
106
107 xiiSimdVec4f center = box.GetCenter();
108 xiiSimdVec4f boxHalfExtents = center - box.m_Min;
109 xiiSimdFloat tmpRadius = boxHalfExtents.GetLength<3>();
110
111 const xiiSimdFloat fSphereRadiusA = (m_CenterAndRadius - center).GetLength<3>() + m_CenterAndRadius.w();
112 const xiiSimdFloat fSphereRadiusB = (rhs.m_CenterAndRadius - center).GetLength<3>() + rhs.m_CenterAndRadius.w();
113
114 m_CenterAndRadius = center;
115 m_CenterAndRadius.SetW(tmpRadius.Min(fSphereRadiusA.Max(fSphereRadiusB)));
116 m_BoxHalfExtents = boxHalfExtents;
117}

124XII_ALWAYS_INLINE void xiiSimdBBoxSphere::Transform(const xiiSimdMat4f& mMat)
125{
126 xiiSimdFloat radius = m_CenterAndRadius.w();
127 m_CenterAndRadius = mMat.TransformPosition(m_CenterAndRadius);
128
129 xiiSimdFloat maxRadius = mMat.m_col0.Dot<3>(mMat.m_col0);
130 maxRadius = maxRadius.Max(mMat.m_col1.Dot<3>(mMat.m_col1));
131 maxRadius = maxRadius.Max(mMat.m_col2.Dot<3>(mMat.m_col2));
132 radius *= maxRadius.GetSqrt();
133
134 m_CenterAndRadius.SetW(radius);
135
136 xiiSimdVec4f newHalfExtents = mMat.m_col0.Abs() * m_BoxHalfExtents.x();
137 newHalfExtents += mMat.m_col1.Abs() * m_BoxHalfExtents.y();
138 newHalfExtents += mMat.m_col2.Abs() * m_BoxHalfExtents.z();
139
140 m_BoxHalfExtents = newHalfExtents.CompMin(xiiSimdVec4f(radius));
141}
