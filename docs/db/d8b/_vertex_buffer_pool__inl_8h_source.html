18template <typename VertexType, typename MutexType, typename AllocatorWrapper>
19XII_ALWAYS_INLINE typename xiiGALVertexBufferPool<VertexType, MutexType, AllocatorWrapper>::AllocationHandle xiiGALVertexBufferPool<VertexType, MutexType, AllocatorWrapper>::Allocate(xiiUInt32 uiCount)
20{
21 XII_ASSERT_DEV(uiCount > 0U, "Allocation size must be greater than zero.");
22
23 XII_LOCK(m_Mutex);
24
25 xiiUInt32 uiChunkIndex = 0U;
26 for (; uiChunkIndex < m_Chunks.GetCount(); ++uiChunkIndex)
27 {
28 const Chunk& chunk = m_Chunks[uiChunkIndex];
29 if (chunk.GetCapacity() - chunk.m_uiUsageCount >= uiCount)
30 break;
31 }
32
33 if (uiChunkIndex == m_Chunks.GetCount())
34 {
35 const xiiUInt64 uiExpandedCapacity = m_Chunks.IsEmpty() ? m_uiInitialChunkSize : m_Chunks.PeekBack().GetCapacity() * m_uiExpansionFactor;
36 const xiiUInt32 uiNewCapacity = static_cast<xiiUInt32>(xiiMath::Max<xiiUInt64>(uiCount, uiExpandedCapacity));
37 m_Chunks.PushBack(Chunk(&m_Allocator, uiNewCapacity, m_pDevice));
38 }
39
40 Chunk& chunk = m_Chunks[uiChunkIndex];
41
42 AllocationHandle handle;
43 handle.m_uiChunkIndex = uiChunkIndex;
44 handle.m_uiOffset = static_cast<xiiUInt32>(chunk.m_uiUsageCount);
45 handle.m_uiCount = uiCount;
46
47 chunk.m_uiUsageCount += uiCount;
48 m_Allocations.PushBack(handle);
49 return handle;
50}

52template <typename VertexType, typename MutexType, typename AllocatorWrapper>
53XII_ALWAYS_INLINE void xiiGALVertexBufferPool<VertexType, MutexType, AllocatorWrapper>::Update(const AllocationHandle& handle, xiiArrayPtr<const VertexType> pData, xiiSharedPtr<xiiGALCommandList> pCommandList)
54{
55 XII_ASSERT_DEV(pData.GetCount() == handle.m_uiCount, "The data size does not match the allocation count.");
56
57 XII_LOCK(m_Mutex);
58
59 XII_ASSERT_DEV(handle.m_uiChunkIndex < m_Chunks.GetCount(), "Invalid chunk index in allocation handle.");
60
61 auto& chunk = m_Chunks[handle.m_uiChunkIndex];
62
63 XII_ASSERT_DEV(handle.m_uiOffset + handle.m_uiCount <= chunk.m_uiUsageCount, "Chunk allocation handle exceeds the used size of the chunk.");
64
65 // Copy data into the CPU-side storage.
66 xiiMemoryUtils::Copy(chunk.m_Vertices.GetData() + handle.m_uiOffset, pData.GetPtr(), pData.GetCount());
67
68 XII_ASSERT_DEV(pCommandList != nullptr, "A valid command list is required to update the GPU buffer.");
69 const xiiArrayPtr<const xiiUInt8> sourceBytes(reinterpret_cast<const xiiUInt8*>(pData.GetPtr()), pData.GetCount() * sizeof(VertexType));
70 pCommandList->UpdateBuffer(chunk.m_pBuffer.Borrow(), handle.m_uiOffset * sizeof(VertexType), sourceBytes);
71}
