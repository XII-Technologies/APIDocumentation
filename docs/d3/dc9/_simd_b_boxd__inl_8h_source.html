70inline void xiiSimdBBoxd::ExpandToInclude(const xiiSimdVec4d* pPoints, xiiUInt32 uiNumPoints, xiiUInt32 uiStride)
71{
72 XII_ASSERT_DEBUG(pPoints != nullptr, "Array may not be nullptr.");
73 XII_ASSERT_DEBUG(uiStride >= sizeof(xiiSimdVec4d), "Data may not overlap.");
74
75 const xiiSimdVec4d* pCur = pPoints;
76
77 for (xiiUInt32 i = 0; i < uiNumPoints; ++i)
78 {
79 ExpandToInclude(*pCur);
80
81 pCur = xiiMemoryUtils::AddByteOffset(pCur, uiStride);
82 }
83}

144XII_ALWAYS_INLINE void xiiSimdBBoxd::Transform(const xiiSimdMat4d& mMat)
145{
146 const xiiSimdVec4d center = GetCenter();
147 const xiiSimdVec4d halfExtents = center - m_Min;
148
149 const xiiSimdVec4d newCenter = mMat.TransformPosition(center);
150
151 xiiSimdVec4d newHalfExtents = mMat.m_col0.Abs() * halfExtents.x();
152 newHalfExtents += mMat.m_col1.Abs() * halfExtents.y();
153 newHalfExtents += mMat.m_col2.Abs() * halfExtents.z();
154
155 *this = xiiSimdBBoxd::MakeFromCenterAndHalfExtents(newCenter, newHalfExtents);
156}
