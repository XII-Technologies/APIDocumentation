93template <typename Class, typename Type>
94class xiiResourceAccessorProperty : public xiiTypedMemberProperty<xiiStringView>
95{
96public:
97 using RealType = xiiStringView;
98 using HandleType = typename xiiTypeTraits<Type>::NonConstReferenceType;
99 using ResourceType = typename HandleType::ResourceType;
100 using GetterFunc = Type (Class::*)() const;
101 using SetterFunc = void (Class::*)(Type value);
102
103 xiiResourceAccessorProperty(xiiStringView sPropertyName, GetterFunc getter, SetterFunc setter) :
104 xiiTypedMemberProperty<RealType>(sPropertyName)
105 {
106 XII_ASSERT_DEBUG(getter != nullptr, "The getter of a property cannot be nullptr.");
107
108 m_Getter = getter;
109 m_Setter = setter;
110
111 if (m_Setter == nullptr)
112 xiiAbstractMemberProperty::m_Flags.Add(xiiPropertyFlags::ReadOnly);
113 }
114
115 virtual void* GetPropertyPointer(const void* pInstance) const override
116 {
117 XII_IGNORE_UNUSED(pInstance);
118
119 // No access to sub-properties, if we have accessors for this property
120 return nullptr;
121 }
122
123 virtual RealType GetValue(const void* pInstance) const override // [tested]
124 {
125 return (static_cast<const Class*>(pInstance)->*m_Getter)().GetResourceID();
126 }
127
128 virtual void SetValue(void* pInstance, RealType value) const override // [tested]
129 {
130 XII_ASSERT_DEV(m_Setter != nullptr, "The property '{0}' has no setter function, thus it is read-only.", xiiAbstractProperty::GetPropertyName());
131
132 if (m_Setter)
133 {
134 if (!value.IsEmpty())
135 {
136 (static_cast<Class*>(pInstance)->*m_Setter)(xiiResourceManager::LoadResource<ResourceType>(value));
137 }
138 else
139 {
140 (static_cast<Class*>(pInstance)->*m_Setter)({});
141 }
142 }
143 }
144
145private:
146 GetterFunc m_Getter;
147 SetterFunc m_Setter;
148};
