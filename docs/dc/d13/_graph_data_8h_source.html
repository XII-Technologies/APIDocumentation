78 void OnGraphEvent(const xiiGraphEvent& event)
79 {
80 if (event.m_Type == xiiGraphEvent::Type::VertexRemoved)
81 m_VertexData.Remove(event.m_hVertex);
82 else if (event.m_Type == xiiGraphEvent::Type::EdgeRemoved)
83 m_EdgeData.Remove(event.m_hEdge);
84 else if (event.m_Type == xiiGraphEvent::Type::BeforeReset)
85 {
86 if (!event.m_pMapping)
87 {
88 m_VertexData.Clear();
89 m_EdgeData.Clear();
90 return;
91 }
92 xiiHashTable<xiiGraphVertexHandle, VertexData> vertices(m_VertexData.GetAllocator());
93 xiiHashTable<xiiGraphEdgeHandle, EdgeData> edges(m_EdgeData.GetAllocator());
94 for (auto it = m_VertexData.GetIterator(); it.IsValid(); ++it)
95 {
96 xiiGraphVertexHandle remapped;
97 if (event.m_pMapping->m_Vertices.TryGetValue(it.Key(), remapped))
98 vertices.Insert(remapped, std::move(it.Value()));
99 }
100 for (auto it = m_EdgeData.GetIterator(); it.IsValid(); ++it)
101 {
102 xiiGraphEdgeHandle remapped;
103 if (event.m_pMapping->m_Edges.TryGetValue(it.Key(), remapped))
104 edges.Insert(remapped, std::move(it.Value()));
105 }
106 m_VertexData = std::move(vertices);
107 m_EdgeData = std::move(edges);
108 }
109 }
110 const xiiGraph& m_Graph;
111 xiiHashTable<xiiGraphVertexHandle, VertexData> m_VertexData;
112 xiiHashTable<xiiGraphEdgeHandle, EdgeData> m_EdgeData;
113 typename xiiGraph::Event::Unsubscriber m_Unsubscriber;
114};
