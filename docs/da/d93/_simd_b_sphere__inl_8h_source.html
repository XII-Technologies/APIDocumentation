33inline xiiSimdBSphere xiiSimdBSphere::MakeFromPoints(const xiiSimdVec4f* pPoints, xiiUInt32 uiNumPoints, xiiUInt32 uiStride /*= sizeof(xiiSimdVec4f)*/)
34{
35 XII_ASSERT_DEBUG(pPoints != nullptr, "The array must not be empty.");
36 XII_ASSERT_DEBUG(uiStride >= sizeof(xiiSimdVec4f), "The data must not overlap.");
37 XII_ASSERT_DEBUG(uiNumPoints > 0, "The array must contain at least one point.");
38
39 xiiSimdBSphere res;
40
41 const xiiSimdVec4f* pCur = pPoints;
42
43 xiiSimdVec4f vCenter = xiiSimdVec4f::MakeZero();
44 for (xiiUInt32 i = 0; i < uiNumPoints; ++i)
45 {
46 vCenter += *pCur;
47 pCur = xiiMemoryUtils::AddByteOffset(pCur, uiStride);
48 }
49
50 res.m_CenterAndRadius = vCenter / xiiSimdFloat(uiNumPoints);
51
52 pCur = pPoints;
53
54 xiiSimdFloat fMaxDistSquare = xiiSimdFloat::MakeZero();
55 for (xiiUInt32 i = 0; i < uiNumPoints; ++i)
56 {
57 const xiiSimdFloat fDistSQR = (*pCur - res.m_CenterAndRadius).GetLengthSquared<3>();
58 fMaxDistSquare = fMaxDistSquare.Max(fDistSQR);
59
60 pCur = xiiMemoryUtils::AddByteOffset(pCur, uiStride);
61 }
62
63 res.m_CenterAndRadius.SetW(fMaxDistSquare.GetSqrt());
64
65 return res;
66}

130inline void xiiSimdBSphere::Transform(const xiiSimdMat4f& mMat)
131{
132 xiiSimdFloat radius = m_CenterAndRadius.w();
133 m_CenterAndRadius = mMat.TransformPosition(m_CenterAndRadius);
134
135 xiiSimdFloat maxRadius = mMat.m_col0.Dot<3>(mMat.m_col0);
136 maxRadius = maxRadius.Max(mMat.m_col1.Dot<3>(mMat.m_col1));
137 maxRadius = maxRadius.Max(mMat.m_col2.Dot<3>(mMat.m_col2));
138 radius *= maxRadius.GetSqrt();
139
140 m_CenterAndRadius.SetW(radius);
141}
