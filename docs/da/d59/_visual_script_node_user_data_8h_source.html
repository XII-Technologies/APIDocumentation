235 {
236 xiiUInt32 m_uiInputArgsMask = 0;
237 xiiUInt32 m_uiOutputArgsMask = 0;
238
239 static xiiResult Serialize(const xiiVisualScriptNodeDescription& nodeDesc, xiiStreamWriter& inout_stream, xiiUInt32& out_uiSize, xiiUInt32& out_uiAlignment)
240 {
241 XII_SUCCEED_OR_RETURN(NodeUserData_Type::Serialize(nodeDesc, inout_stream, out_uiSize, out_uiAlignment));
242
243 const xiiVariantArray& propertiesVar = nodeDesc.m_Value.Get<xiiVariantArray>();
244 XII_ASSERT_DEBUG(propertiesVar.GetCount() == 1, "Invalid number of properties");
245
246 xiiHashedString sFunctionName = propertiesVar[0].Get<xiiHashedString>();
247 inout_stream << sFunctionName;
248
249 const xiiRTTI* pType = xiiRTTI::FindTypeByName(nodeDesc.m_sTargetTypeName);
250 if (pType == nullptr)
251 return XII_FAILURE;
252
253 const xiiAbstractFunctionProperty* pFunction = nullptr;
254 for (auto pFunc : pType->GetFunctions())
255 {
256 if (pFunc->GetPropertyName() == sFunctionName)
257 {
258 pFunction = pFunc;
259 break;
260 }
261 }
262
263 if (pFunction == nullptr)
264 return XII_FAILURE;
265
266 auto pScriptableFunctionAttribute = pFunction->GetAttributeByType<xiiScriptableFunctionAttribute>();
267 if (pScriptableFunctionAttribute == nullptr)
268 return XII_FAILURE;
269
270 xiiUInt32 uiInputArgsMask = 0;
271 xiiUInt32 uiOutputArgsMask = 0;
272 for (xiiUInt32 i = 0; i < pScriptableFunctionAttribute->GetArgumentCount(); ++i)
273 {
274 auto argType = pScriptableFunctionAttribute->GetArgumentType(i);
275 if (argType == xiiScriptableFunctionAttribute::In || argType == xiiScriptableFunctionAttribute::Inout)
276 {
277 uiInputArgsMask |= XII_BIT(i);
278 }
279
280 if (argType == xiiScriptableFunctionAttribute::Out || argType == xiiScriptableFunctionAttribute::Inout)
281 {
282 uiOutputArgsMask |= XII_BIT(i);
283 }
284 }
285
286 inout_stream << uiInputArgsMask;
287 inout_stream << uiOutputArgsMask;
288
289 out_uiSize = sizeof(NodeUserData_TypeAndFunction);
290 out_uiAlignment = GetUserDataAlignment<NodeUserData_TypeAndFunction>();
291 return XII_SUCCESS;
292 }

294 static xiiResult Deserialize(xiiVisualScriptGraphDescription::Node& ref_node, xiiStreamReader& inout_stream, xiiUInt8*& inout_pAdditionalData)
295 {
296 auto& userData = ref_node.InitUserData<NodeUserData_TypeAndFunction>(inout_pAdditionalData);
297 XII_SUCCEED_OR_RETURN(ReadType(inout_stream, userData.m_pType));
298 XII_SUCCEED_OR_RETURN(ReadProperty(inout_stream, userData.m_pType, userData.m_pType->GetFunctions(), userData.m_pProperty));
299
300 inout_stream >> userData.m_uiInputArgsMask;
301 inout_stream >> userData.m_uiOutputArgsMask;
302
303 if (static_cast<const xiiAbstractFunctionProperty*>(userData.m_pProperty)->GetArgumentCount() != xiiMath::CountBits(userData.m_uiInputArgsMask | userData.m_uiOutputArgsMask))
304 {
305 xiiLog::Error("Visual script {} '{}': Argument count mismatch. Script needs re-transform.", xiiVisualScriptNodeDescription::Type::GetName(ref_node.m_Type), userData.m_pProperty->GetPropertyName());
306 return XII_FAILURE;
307 }
308
309 return XII_SUCCESS;
310 }
