10xiiResult xiiScreen::EnumerateScreens(xiiHybridArray<xiiScreenInfo, 2>& out_Screens)
11{
12 out_Screens.Clear();
13
14 if (!SDL_Init(SDL_INIT_VIDEO))
15 {
16 xiiLog::Error("Unable to initialize SDL Video: {}", SDL_GetError());
17 return XII_FAILURE;
18 }
19
20 xiiInt32 iDisplayCount;
21 SDL_DisplayID* pDisplayIDs = SDL_GetDisplays(&iDisplayCount);
22
23 XII_SCOPE_EXIT(if (pDisplayIDs != nullptr) { SDL_free(pDisplayIDs); });
24
25 out_Screens.Reserve(iDisplayCount);
26
27 for (xiiInt32 i = 0; i < iDisplayCount; ++i)
28 {
29 xiiScreenInfo& screen = out_Screens.ExpandAndGetRef();
30 screen.m_sDisplayName = SDL_GetDisplayName(pDisplayIDs[i]);
31
32 if (screen.m_sDisplayName.IsEmpty())
33 {
34 xiiLog::Error("SDL failed to retrieve display name for display {} with error '{}'.", i, SDL_GetError());
35 return XII_FAILURE;
36 }
37
38 SDL_Rect displayBounds;
39 if (!SDL_GetDisplayBounds(pDisplayIDs[i], &displayBounds))
40 {
41 xiiLog::Error("SDL failed to retrieve display bounds for display {} with error '{}'.", i, SDL_GetError());
42 return XII_FAILURE;
43 }
44
45 // SDL reports screen coordinates from top/left to bottom/right.
46 screen.m_iOffsetX = displayBounds.x;
47 screen.m_iOffsetY = displayBounds.y;
48 screen.m_iResolutionX = displayBounds.w;
49 screen.m_iResolutionY = displayBounds.h;
50 screen.m_bIsPrimary = i == 0; // Primary Monitor at index 0
51 }
52
53 if (out_Screens.IsEmpty())
54 return XII_FAILURE;
55
56 return XII_SUCCESS;
57}
