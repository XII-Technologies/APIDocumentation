12 // this is the size of the temp buffer that BuildString functions get for writing their result to.
13 // The buffer is always available and allocated on the stack, so this prevents the need for memory allocations.
14 // If a BuildString function requires no storage at all, it can return a xiiStringView to unrelated memory
15 // (e.g. if the memory already exists).
16 // If a BuildString function requires more storage, it may need to do some trickery.
17 // For an example look at BuildString for xiiArgErrorCode, which uses an increased thread_local temp buffer.
18 static constexpr xiiUInt32 TempStringLength = 64;
19 // Maximum number of parameters. Results in compilation error if exceeded.
20 static constexpr xiiUInt32 MaxNumParameters = 12;

65 template <xiiInt32 N>
66 typename std::enable_if<sizeof...(ARGS) != N>::type ReplaceString(char tmp[MaxNumParameters][TempStringLength], xiiStringView* pViews) const
67 {
68 static_assert(N < MaxNumParameters, "Maximum number of format arguments reached");
69
70 // using a free function allows to overload with various different argument types
71 pViews[N] = BuildString(tmp[N], TempStringLength - 1, std::get<N>(m_Arguments));
72
73 // Recurse, chip off one argument
74 ReplaceString<N + 1>(tmp, pViews);
75 }
76
77 // Recursion end if we reached the number of arguments.
78 template <xiiInt32 N>
79 typename std::enable_if<sizeof...(ARGS) == N>::type ReplaceString(char tmp[MaxNumParameters][TempStringLength], xiiStringView* pViews) const
80 {
81 XII_IGNORE_UNUSED(tmp);
82 XII_IGNORE_UNUSED(pViews);
83 }
84
85 // stores the arguments
86 std::tuple<ARGS...> m_Arguments;
