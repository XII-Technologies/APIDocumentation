52template <typename Type, xiiCVarType::Enum CVarType>
53void xiiTypedCVar<Type, CVarType>::operator=(const Type& value)
54{
55 xiiCVarEvent e(this);
56
57 if (GetFlags().IsAnySet(xiiCVarFlags::RequiresDelayedSync))
58 {
59 if (value == m_Values[xiiCVarValue::DelayedSync]) // No change
60 return;
61
62 e.m_EventType = xiiCVarEvent::DelayedSyncValueChanged;
63 }
64 else
65 {
66 if (m_Values[xiiCVarValue::Current] == value) // No change
67 return;
68
69 m_Values[xiiCVarValue::Current] = value;
70 e.m_EventType = xiiCVarEvent::ValueChanged;
71 }
72
73 m_Values[xiiCVarValue::DelayedSync] = value;
74
75 m_CVarEvents.Broadcast(e);
76
77 // Broadcast the same to the 'all cvars' event handlers.
78 s_AllCVarEvents.Broadcast(e);
79}
