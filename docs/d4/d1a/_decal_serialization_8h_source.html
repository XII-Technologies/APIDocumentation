12 // These descriptor formats do not use string deduplication. Bound metadata
13 // before allocation and require complete UTF-8 before interning resource IDs.
14 inline xiiResult ReadString(xiiStreamReader& stream, xiiStringBuilder& text)
15 {
16 xiiUInt32 length = 0;
17 if (stream.ReadDWordValue(&length).Failed() || length > MaxStringBytes)
18 return XII_FAILURE;
19 xiiHybridArray<char, 256> bytes;
20 bytes.SetCountUninitialized(length + 1);
21 if (stream.ReadBytes(bytes.GetData(), length) != length)
22 return XII_FAILURE;
23 bytes[length] = '\0';
24 if (!xiiUnicodeUtils::IsValidUtf8(bytes.GetData(), bytes.GetData() + length) || xiiStringUtils::GetStringElementCount(bytes.GetData()) != length)
25 return XII_FAILURE;
26 text = xiiStringView(bytes.GetData(), length);
27 return XII_SUCCESS;
28 }

30 template <typename ResourceType>
31 xiiResult ReadResourceHandle(xiiStreamReader& stream, xiiTypedResourceHandle<ResourceType>& handle)
32 {
33 xiiStringBuilder type, id;
34 XII_SUCCEED_OR_RETURN(ReadString(stream, type));
35 if (type.IsEmpty())
36 {
37 handle.Invalidate();
38 return XII_SUCCESS;
39 }
40 XII_SUCCEED_OR_RETURN(ReadString(stream, id));
41 const auto* rtti = xiiResourceManager::FindResourceForAssetType(type);
42 if (rtti == nullptr)
43 rtti = xiiRTTI::FindTypeByName(type);
44 if (id.IsEmpty() || rtti != xiiGetStaticRTTI<ResourceType>())
45 return XII_FAILURE;
46 handle = xiiResourceManager::LoadResource<ResourceType>(id);
47 return XII_SUCCESS;
48 }
