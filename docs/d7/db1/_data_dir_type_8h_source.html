194 virtual xiiUInt64 Read(void* pBuffer, xiiUInt64 uiBytes) = 0;
195
197 virtual xiiUInt64 Skip(xiiUInt64 uiBytes)
198 {
199 xiiUInt8 uiTempBuffer[1024];
200
201 xiiUInt64 uiBytesSkipped = 0;
202
203 while (uiBytesSkipped < uiBytes)
204 {
205 xiiUInt64 uiBytesToRead = xiiMath::Min<xiiUInt64>(uiBytes - uiBytesSkipped, 1024);
206
207 xiiUInt64 uiBytesRead = Read(uiTempBuffer, uiBytesToRead);
208
209 uiBytesSkipped += uiBytesRead;
210
211 // Terminate early if the stream didn't read as many bytes as we requested (EOF for example)
212 if (uiBytesRead < uiBytesToRead)
213 break;
214 }
215
216 return uiBytesSkipped;
217 }
218};
