195 virtual xiiUInt64 Skip(xiiUInt64 uiBytes)
196 {
197 xiiUInt8 uiTempBuffer[1024];
198
199 xiiUInt64 uiBytesSkipped = 0;
200
201 while (uiBytesSkipped < uiBytes)
202 {
203 xiiUInt64 uiBytesToRead = xiiMath::Min<xiiUInt64>(uiBytes - uiBytesSkipped, 1024);
204
205 xiiUInt64 uiBytesRead = Read(uiTempBuffer, uiBytesToRead);
206
207 uiBytesSkipped += uiBytesRead;
208
209 // Terminate early if the stream didn't read as many bytes as we requested (EOF for example)
210 if (uiBytesRead < uiBytesToRead)
211 break;
212 }
213
214 return uiBytesSkipped;
215 }
