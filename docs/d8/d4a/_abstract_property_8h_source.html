94 template <class Type>
95 static xiiBitflags<xiiPropertyFlags> GetParameterFlags()
96 {
97 using CleanType = typename xiiTypeTraits<Type>::NonConstReferencePointerType;
98
99 xiiBitflags<xiiPropertyFlags> flags;
100 constexpr xiiVariantType::Enum type = static_cast<xiiVariantType::Enum>(xiiVariantTypeDeduction<CleanType>::value);
101 if constexpr (std::is_same<CleanType, xiiVariant>::value || (type >= xiiVariantType::FirstStandardType && type <= xiiVariantType::LastStandardType) || std::is_same<Type, const char*>::value) // We treat const char* as a basic type and not a pointer.
102 flags.Add(xiiPropertyFlags::StandardType);
103 else if constexpr (xiiIsEnum<CleanType>::value)
104 flags.Add(xiiPropertyFlags::IsEnum);
105 else if constexpr (xiiIsBitflags<CleanType>::value)
106 flags.Add(xiiPropertyFlags::Bitflags);
107 else
108 flags.Add(xiiPropertyFlags::Class);
109
110 if constexpr (std::is_const<typename xiiTypeTraits<Type>::NonReferencePointerType>::value)
111 flags.Add(xiiPropertyFlags::Const);
112
113 if constexpr (std::is_pointer<Type>::value && !std::is_same<Type, const char*>::value)
114 flags.Add(xiiPropertyFlags::Pointer);
115
116 if constexpr (std::is_reference<Type>::value)
117 flags.Add(xiiPropertyFlags::Reference);
118
119 return flags;
120 }
121};
