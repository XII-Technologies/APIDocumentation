21template <typename Element>
22struct DefaultRadixKeyExtractor
23{
24 constexpr xiiUInt64 GetKey(const Element& value) const
25 {
26 if constexpr (std::is_unsigned_v<Element>)
27 {
28 return static_cast<xiiUInt64>(value);
29 }
30 else if constexpr (std::is_signed_v<Element>)
31 {
32 using U = std::make_unsigned_t<Element>;
33 constexpr U SignBit = U(1) << (sizeof(Element) * 8 - 1);
34
35 return static_cast<xiiUInt64>(static_cast<U>(value) ^ SignBit);
36 }
37 else if constexpr (std::is_enum_v<Element>)
38 {
39 using Under = std::underlying_type_t<Element>;
40
41 return DefaultRadixKeyExtractor<Under>{}.GetKey(static_cast<Under>(value));
42 }
43 else
44 {
45 static_assert(std::is_integral_v<Element>, "DefaultRadixKeyExtractor only supports integral or enum types. Provide a custom key extractor.");
46 }
47 }
48};

51template <typename Element, typename KeyFunc>
52constexpr xiiUInt64 ExtractRadixKey(const KeyFunc& keyFunc, const Element& value)
53{
54 if constexpr (HasGetKey<KeyFunc, Element>)
55 {
56 return static_cast<xiiUInt64>(keyFunc.GetKey(value));
57 }
58 else if constexpr (CallableKey<KeyFunc, Element>)
59 {
60 return static_cast<xiiUInt64>(keyFunc(value));
61 }
62 else
63 {
64 static_assert(HasGetKey<KeyFunc, Element> || CallableKey<KeyFunc, Element>, "KeyFunc must provide either GetKey(Element) or operator()(Element).");
65 }
66}
