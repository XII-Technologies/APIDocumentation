12 inline bool FloatReadable(xiiGALResourceFormat::Enum format)
13 {
14 const auto& description = xiiGALTextureUtilities::GetResourceFormatProperties(format);
15 const auto type = description.m_ComponentType;
16 return !description.IsTypeless() && (type == xiiGALResourceFormatComponentType::Float || description.IsNormalized() || type == xiiGALResourceFormatComponentType::UnsignedNormalizedSRGB || type == xiiGALResourceFormatComponentType::Depth || type == xiiGALResourceFormatComponentType::DepthStencil);

24 inline xiiResult Validate(xiiRenderPipelinePassContext& context, Inputs& data, bool requireEmission = true)
25 {
26 const auto* depth = context.m_Graph.GetTextureDescription(data.m_Depth);
27 const auto* camera = context.m_Graph.GetBufferDescription(data.m_Camera);
28 if (!depth || !depth->m_Size.HasNonZeroArea() || !camera || camera->m_uiSize != sizeof(xiiFrameCameraConstants))
29 return context.ReportFailure("Surface lighting requires nonempty depth and exact camera constants.");
30 data.m_Size = depth->m_Size;
31 const auto compatible = [&](xiiRenderGraphTextureHandle handle, xiiUInt32 channels) {
32 const auto* d = context.m_Graph.GetTextureDescription(handle);
33 return d && d->m_Size == data.m_Size && d->m_Type == xiiGALResourceDimension::Texture2D && d->m_uiArraySizeOrDepth == 1 && d->m_uiSampleCount == 1 && d->m_BindFlags.IsSet(xiiGALBindFlags::ShaderResource) && FloatReadable(d->m_Format) && xiiGALTextureUtilities::GetComponentCount(d->m_Format) >= channels;
34 };
35 if (!compatible(data.m_Depth, 1) || !compatible(data.m_Normal, 2) || !compatible(data.m_Albedo, 4) || !compatible(data.m_Material, 4) || (requireEmission && !compatible(data.m_Emissive, 4)))
36 return context.ReportFailure("Surface images must have matching extents, sample count and shader-readable channel layouts.");
37 return XII_SUCCESS;
38 }
39 inline Inputs Read(const Inputs& source, xiiRenderGraphBuilder& builder)
40 {
41 auto data = source;
42 data.m_Depth = builder.ReadTexture(source.m_Depth, xiiGALResourceStateFlags::ShaderResource);
43 data.m_Normal = builder.ReadTexture(source.m_Normal, xiiGALResourceStateFlags::ShaderResource);
44 data.m_Albedo = builder.ReadTexture(source.m_Albedo, xiiGALResourceStateFlags::ShaderResource);
45 data.m_Material = builder.ReadTexture(source.m_Material, xiiGALResourceStateFlags::ShaderResource);
46 if (source.m_Emissive.IsValid())
47 data.m_Emissive = builder.ReadTexture(source.m_Emissive, xiiGALResourceStateFlags::ShaderResource);
48 data.m_Camera = builder.ReadBuffer(source.m_Camera, xiiGALResourceStateFlags::ConstantBuffer);
49 return data;
50 }
51 inline void Bind(const Inputs& data, xiiRenderGraphPassContext& execute)
52 {
53 auto& commands = execute.GetCommandList();
54 commands.ResolveAndSetConstantBuffer("xiiSurfaceCameraConstants", execute.GetBuffer(data.m_Camera), xiiGALShaderType::Compute);
55 commands.ResolveAndSetShaderResourceTextureView("g_SurfaceDepth", execute.GetTexture(data.m_Depth)->GetDefaultView(xiiGALTextureViewType::ShaderResource), xiiGALShaderType::Compute);
56 commands.ResolveAndSetShaderResourceTextureView("g_SurfaceNormal", execute.GetTexture(data.m_Normal)->GetDefaultView(xiiGALTextureViewType::ShaderResource), xiiGALShaderType::Compute);
57 commands.ResolveAndSetShaderResourceTextureView("g_SurfaceAlbedo", execute.GetTexture(data.m_Albedo)->GetDefaultView(xiiGALTextureViewType::ShaderResource), xiiGALShaderType::Compute);
58 commands.ResolveAndSetShaderResourceTextureView("g_SurfaceMaterial", execute.GetTexture(data.m_Material)->GetDefaultView(xiiGALTextureViewType::ShaderResource), xiiGALShaderType::Compute);
59 if (data.m_Emissive.IsValid())
60 commands.ResolveAndSetShaderResourceTextureView("g_SurfaceEmissive", execute.GetTexture(data.m_Emissive)->GetDefaultView(xiiGALTextureViewType::ShaderResource), xiiGALShaderType::Compute);
61 }
62 inline xiiRenderGraphTextureHandle Output(xiiRenderGraphBuilder& builder, xiiSizeU32 size)
63 {
64 xiiGALTextureCreationDescription d;
65 d.m_Type = xiiGALResourceDimension::Texture2D;
66 d.m_Size = size;
67 d.m_Format = xiiGALResourceFormat::RGBA16Float;
68 d.m_BindFlags = xiiGALBindFlags::ShaderResource | xiiGALBindFlags::UnorderedAccess;

70 }
71 inline void BindConstants(xiiRenderGraphPassContext& execute, xiiRenderGraphBufferHandle buffer, const xiiSurfaceLightingConstants& snapshot)
72 {
73 auto& commands = execute.GetCommandList();
74 {
75 xiiGALMapHelper<xiiSurfaceLightingConstants> mapped(commands, execute.GetBuffer(buffer), xiiGALMapType::Write, xiiGALMapFlags::Discard);
76 *mapped = snapshot;
77 }
78 xiiGALStateTransitionDescription transition;
79 transition.m_pResource = execute.GetBuffer(buffer);
80 transition.m_NewState = xiiGALResourceStateFlags::ConstantBuffer;
81 transition.m_TransitionFlags = xiiGALStateTransitionFlags::UpdateState;
82 commands.TransitionResourceStates(xiiMakeArrayPtr(&transition, 1));
83 commands.ResolveAndSetConstantBuffer("xiiSurfaceLightingConstants", execute.GetBuffer(buffer), xiiGALShaderType::Compute);
84 }
