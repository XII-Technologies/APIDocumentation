14 // Resolve OLD_NAME-only case changes while the object exists, or recognize a delayed record against the mirror even if the renamed object was deleted.
15 bool ReconcileCaseRename(xiiFileSystemMirror<bool>& mirror, xiiStringView oldPath, xiiString& out_sNewPath, xiiDirectoryWatcher::EnumerateChangesFunction callback)
16 {
17 xiiStringBuilder sCached, sCorrected;
18 xiiFileSystemMirror<bool>::Type type;
19
20 if (mirror.GetPathCasing(oldPath, sCached, type).Failed())
21 return false;
22
23 xiiFileStats stats;
24 if (xiiOSFile::GetFileStats(oldPath, stats).Failed())
25 {
26 if (sCached.GetFileNameAndExtension() == oldPath.GetFileNameAndExtension())
27 return false;
28
29 out_sNewPath = sCached;
30 return true;
31 }
32
33 if (!stats.m_sName.IsEqual_NoCase(oldPath.GetFileNameAndExtension()))
34 return false;
35
36 // Repair the leaf in its sCached parent. An ancestor may itself still need repair; creating a second parent here would split the sCached subtree.
37 sCorrected = sCached.GetFileDirectory();
38 sCorrected.AppendPath(stats.m_sName);
39
40 // A normal rename followed by replacement can leave a different file at the old path
41 // before its notifications are consumed. An unchanged leaf does not prove a case rename.
42 if (sCached == sCorrected && sCached.GetFileNameAndExtension() == oldPath.GetFileNameAndExtension())
43 return false;
44
45 if (sCached != sCorrected)
46 {
47 const bool bIsDirectory = type == xiiFileSystemMirror<bool>::Type::Directory;
48 if (bIsDirectory)
49 {
50 if (mirror.MoveDirectory(sCached, sCorrected).Failed())
51 return false;
52 }
53 else
54 {
55 if (mirror.MoveFile(sCached, sCorrected).Failed())
56 return false;
57 }
58
59 const xiiDirectoryWatcherType eventType = bIsDirectory ? xiiDirectoryWatcherType::Directory : xiiDirectoryWatcherType::File;
60
61 callback(sCached, xiiDirectoryWatcherAction::RenamedOldName, eventType);
62 callback(sCorrected, xiiDirectoryWatcherAction::RenamedNewName, eventType);
63 }
64
65 out_sNewPath = sCorrected;
66 return true;
67 }

69 // ReFS can withhold case-only notifications indefinitely. Bound metadata queries per poll; file contents are never read.
70 // Each batch walks the cached tree, O(N), and checks at most 128 paths. The caller limits batches to 100 ms.
71 void ReconcileMirrorPaths(xiiFileSystemMirror<bool>& mirror, xiiStringView root, xiiUInt32& nextEntry, bool bIsRecursive, xiiDirectoryWatcher::EnumerateChangesFunction callback)
72 {
73 struct Mutation
74 {
75 xiiString m_sOldPath;
76 xiiString m_sNewPath;
77 xiiDirectoryWatcherType m_Type;
78 bool m_bRemoved = false;
79 };
80
81 xiiTemporaryHybridArray<Mutation, 4> changes;
82 xiiUInt32 uiOrdinal = 0;
83 xiiUInt32 uiChecked = 0;
84 constexpr xiiUInt32 uiBatchSize = 128;
85 const xiiResult result = mirror.Enumerate(root, [&](const xiiStringBuilder& path, xiiFileSystemMirror<bool>::Type type) -> void {
86 if (!bIsRecursive && path.GetFileDirectory().GetElementCount() != root.GetElementCount() + 1)
87 return;
88
89 const xiiUInt32 uiCurrent = uiOrdinal++;
90 if (uiCurrent < nextEntry || uiChecked >= uiBatchSize)
91 return;
92
93 ++uiChecked;
94
95 xiiFileStats stats;
96 if (xiiOSFile::GetFileStats(path, stats).Failed())
97 {
98 // Do not interpret inaccessible objects as deleted. Confirm a missing path using the native error, which the portable stats result omits.
99 const DWORD uiAttributes = GetFileAttributesW(xiiDosDevicePath(path));
100 const DWORD uiError = GetLastError();
101
102 if (uiAttributes == INVALID_FILE_ATTRIBUTES && (uiError == ERROR_FILE_NOT_FOUND || uiError == ERROR_PATH_NOT_FOUND))
103 {
104 changes.PushBack({path, {}, type == xiiFileSystemMirror<bool>::Type::Directory ? xiiDirectoryWatcherType::Directory : xiiDirectoryWatcherType::File, true});
105 }
106 return;
107 }
108
109 if (stats.m_bIsDirectory != (type == xiiFileSystemMirror<bool>::Type::Directory) || path.GetFileNameAndExtension() == stats.m_sName || !path.GetFileNameAndExtension().IsEqual_NoCase(stats.m_sName))
110 return;
111
112 xiiStringBuilder sCorrected = path;
113 sCorrected.PathParentDirectory();
114 sCorrected.AppendPath(stats.m_sName);
115
116 changes.PushBack({path, sCorrected, stats.m_bIsDirectory ? xiiDirectoryWatcherType::Directory : xiiDirectoryWatcherType::File});
117 });
118
119 nextEntry = nextEntry + uiChecked >= uiOrdinal ? 0 : nextEntry + uiChecked;
120
121 if (result.Failed())
122 return;
123
124 // Enumeration is read-only; apply child changes before parent changes after it ends.
125 for (const auto& rename : changes)
126 {
127 if (rename.m_bRemoved)
128 {
129 // A parent can disappear after its children were checked in an earlier batch. Report any remaining descendants before removing the subtree.
130 if (bIsRecursive && rename.m_Type == xiiDirectoryWatcherType::Directory)
131 mirror.Enumerate(rename.m_sOldPath, [&](const xiiStringBuilder& path, xiiFileSystemMirror<bool>::Type type) -> void {
132 callback(path, xiiDirectoryWatcherAction::Removed, type == xiiFileSystemMirror<bool>::Type::Directory ? xiiDirectoryWatcherType::Directory : xiiDirectoryWatcherType::File);
133 })
134 .IgnoreResult();
135
136 const xiiResult removeResult = rename.m_Type == xiiDirectoryWatcherType::Directory ? mirror.RemoveDirectory(rename.m_sOldPath) : mirror.RemoveFile(rename.m_sOldPath);
137 if (removeResult.Succeeded())
138 {
139 callback(rename.m_sOldPath, xiiDirectoryWatcherAction::Removed, rename.m_Type);
140 }
141 continue;
142 }
143
144 const xiiResult moveResult = rename.m_Type == xiiDirectoryWatcherType::Directory ? mirror.MoveDirectory(rename.m_sOldPath, rename.m_sNewPath) : mirror.MoveFile(rename.m_sOldPath, rename.m_sNewPath);
145 if (moveResult.Succeeded())
146 {
147 callback(rename.m_sOldPath, xiiDirectoryWatcherAction::RenamedOldName, rename.m_Type);
148 callback(rename.m_sNewPath, xiiDirectoryWatcherAction::RenamedNewName, rename.m_Type);
149 }
150 }
151 }
