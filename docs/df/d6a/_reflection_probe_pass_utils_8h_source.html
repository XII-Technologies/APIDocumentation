21 inline Bindings Read(const xiiSharedPtr<const xiiReflectionProbeFrameData>& snapshot, xiiRenderGraphBuilder& builder)
22 {
23 Bindings result;
24 result.m_Snapshot = snapshot;
25 if (!snapshot->m_Textures.IsEmpty())
26 {
27 result.m_Records = builder.ReadBuffer(snapshot->m_Records, xiiGALResourceStateFlags::ShaderResource);
28 result.m_Constants = builder.ReadBuffer(snapshot->m_Constants, xiiGALResourceStateFlags::ConstantBuffer);
29 }
30 result.m_SkyConstants = builder.ReadBuffer(snapshot->m_SkyConstants, xiiGALResourceStateFlags::ConstantBuffer);
31 // Bind an already retained local image when the shader's sky branch is disabled.
32 const auto& sky = snapshot->m_Sky ? snapshot->m_Sky : snapshot->m_Environments[0];
33 builder.WaitForFence(sky->m_Ready, 1).AssertSuccess();
34 result.m_SkyRadiance = builder.ReadTexture(builder.ImportTexture("Sky Specular Cubemap", sky->m_Specular, xiiGALResourceStateFlags::Unknown), xiiGALResourceStateFlags::ShaderResource);
35 result.m_SkyIrradiance = builder.ReadTexture(builder.ImportTexture("Sky Irradiance Cubemap", sky->m_Irradiance, xiiGALResourceStateFlags::Unknown), xiiGALResourceStateFlags::ShaderResource);
36 for (xiiUInt32 i = 0; i < snapshot->m_Textures.GetCount(); ++i)
37 {
38 xiiStringBuilder name;
39 name.SetFormat("Local Reflection Cubemap {}", i);
40 result.m_Textures[i] = builder.ReadTexture(builder.ImportTexture(name, snapshot->m_Textures[i], xiiGALResourceStateFlags::Unknown), xiiGALResourceStateFlags::ShaderResource);
41 const auto& environment = snapshot->m_Environments[i];
42 builder.WaitForFence(environment->m_Ready, 1).AssertSuccess();
43 name.SetFormat("Local Irradiance Cubemap {}", i);
44 result.m_Irradiance[i] = builder.ReadTexture(builder.ImportTexture(name, environment->m_Irradiance, xiiGALResourceStateFlags::Unknown), xiiGALResourceStateFlags::ShaderResource);
45 }
46 return result;
47 }

49 inline void Bind(const Bindings& data, xiiRenderGraphPassContext& execute, xiiBitflags<xiiGALShaderType> stages)
50 {
51 auto& commands = execute.GetCommandList();
52 commands.ResolveAndSetConstantBuffer("xiiSkyLightConstants", execute.GetBuffer(data.m_SkyConstants), stages);
53 commands.ResolveAndSetShaderResourceTextureView("g_SkyRadiance", execute.GetTexture(data.m_SkyRadiance)->GetDefaultView(xiiGALTextureViewType::ShaderResource), stages);
54 commands.ResolveAndSetShaderResourceTextureView("g_SkyIrradiance", execute.GetTexture(data.m_SkyIrradiance)->GetDefaultView(xiiGALTextureViewType::ShaderResource), stages);
55 if (data.m_Snapshot->m_Textures.IsEmpty())
56 return;
57 commands.ResolveAndSetConstantBuffer("xiiReflectionProbeConstants", execute.GetBuffer(data.m_Constants), stages);
58 commands.ResolveAndSetShaderResourceBufferView("g_ReflectionProbeData", execute.GetBuffer(data.m_Records)->GetDefaultView(xiiGALBufferViewType::ShaderResource), stages);
59 xiiSharedPtr<xiiGALTextureView> views[XII_MAX_REFLECTION_PROBES];
60 xiiGALTextureView* images[XII_MAX_REFLECTION_PROBES];
61 for (xiiUInt32 i = 0; i < XII_MAX_REFLECTION_PROBES; ++i)
62 {
63 views[i] = execute.GetTexture(data.m_Textures[i < data.m_Snapshot->m_Textures.GetCount() ? i : 0])->GetDefaultView(xiiGALTextureViewType::ShaderResource);
64 images[i] = views[i].Borrow();
65 }
66 commands.ResolveAndSetShaderResourceTextureViews("g_ReflectionProbeTextures", 0, images, stages);
67 for (xiiUInt32 i = 0; i < XII_MAX_REFLECTION_PROBES; ++i)
68 {
69 views[i] = execute.GetTexture(data.m_Irradiance[i < data.m_Snapshot->m_Environments.GetCount() ? i : 0])->GetDefaultView(xiiGALTextureViewType::ShaderResource);
70 images[i] = views[i].Borrow();
71 }
72 commands.ResolveAndSetShaderResourceTextureViews("g_ReflectionProbeIrradiance", 0, images, stages);
73 }
