182 inline float CalculateSphereScreenCoverage(const xiiBoundingSphere& sphere, const xiiVec3& vCameraPosition, xiiAngle perspectiveCameraFov)
183 {
184 const float fDist = (sphere.m_vCenter - vCameraPosition).GetLength();
185 const float fHalfHeight = xiiMath::Tan(perspectiveCameraFov * 0.5f) * fDist;
186 return sphere.m_fRadius / fHalfHeight;
187 }
