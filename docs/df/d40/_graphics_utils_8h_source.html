184 inline float CalculateSphereScreenCoverage(const xiiBoundingSphere& sphere, const xiiVec3& vCameraPosition, xiiAngle perspectiveCameraFov)
185 {
186 const float fDist = (sphere.m_vCenter - vCameraPosition).GetLength();
187 const float fHalfHeight = xiiMath::Tan(perspectiveCameraFov * 0.5f) * fDist;
188 return sphere.m_fRadius / fHalfHeight;
189 }

201 inline float CalculateSphereScreenCoverage(float fSphereRadius, float fOrthoCameraDimensions)
202 {
203 const float fHalfHeight = fOrthoCameraDimensions * 0.5f;
204 return fSphereRadius / fHalfHeight;
205 }
